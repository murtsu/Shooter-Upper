cmake_minimum_required(VERSION 3.16)
project(shooter-upper VERSION 1.0.0 LANGUAGES C)

set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
if(NOT CMAKE_BUILD_TYPE AND NOT CMAKE_CONFIGURATION_TYPES)
  set(CMAKE_BUILD_TYPE Release)
endif()

find_package(SDL2 REQUIRED)
find_library(M_LIB m)

option(SU_SANITIZE "Build with AddressSanitizer and UBSan" OFF)

add_compile_options(-Wall -Wextra -Wshadow)
if(SU_SANITIZE)
  add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
  add_link_options(-fsanitize=address,undefined)
endif()

# the simulation: no SDL in here
add_library(su_game STATIC src/game/game.c)
target_link_libraries(su_game PUBLIC ${M_LIB})

add_executable(shooter-upper
  src/main.c src/gfx.c src/audio.c src/music.c src/input.c src/storage.c
  src/sense.c src/errors.c src/cast.c)
target_link_libraries(shooter-upper PRIVATE su_game SDL2::SDL2 ${M_LIB})
target_compile_definitions(shooter-upper PRIVATE _POSIX_C_SOURCE=200809L)

include(CTest)
add_executable(test_game tests/test_game.c)
target_link_libraries(test_game PRIVATE su_game)
add_test(NAME game COMMAND test_game)

add_executable(test_audio tests/test_audio.c src/audio.c src/music.c)
target_link_libraries(test_audio PRIVATE SDL2::SDL2 ${M_LIB})
add_test(NAME audio COMMAND test_audio)

add_executable(test_music tests/test_music.c src/music.c src/audio.c)
target_link_libraries(test_music PRIVATE SDL2::SDL2 ${M_LIB})
add_test(NAME music COMMAND test_music)

add_executable(test_sense tests/test_sense.c src/sense.c)
target_link_libraries(test_sense PRIVATE ${M_LIB})
add_test(NAME sense COMMAND test_sense)

add_executable(test_errors tests/test_errors.c src/errors.c)
add_test(NAME errors COMMAND test_errors)

add_executable(test_cast tests/test_cast.c src/cast.c src/gfx.c)
target_link_libraries(test_cast PRIVATE su_game ${M_LIB})
add_test(NAME cast COMMAND test_cast)

add_test(NAME selftest COMMAND shooter-upper --selftest --config-dir ${CMAKE_BINARY_DIR}/selftest-cfg)
set_tests_properties(selftest PROPERTIES ENVIRONMENT "SDL_VIDEODRIVER=dummy;SDL_AUDIODRIVER=dummy")
add_test(NAME smoke COMMAND shooter-upper --smoke 7200 --config-dir ${CMAKE_BINARY_DIR}/smoke-cfg)

include(GNUInstallDirs)
install(TARGETS shooter-upper RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
install(FILES packaging/shooter-upper.desktop DESTINATION ${CMAKE_INSTALL_DATADIR}/applications)
install(FILES packaging/shooter-upper.svg DESTINATION ${CMAKE_INSTALL_DATADIR}/icons/hicolor/scalable/apps)
# the song that ships with the game, if there is one in music/theme.mp3
if(EXISTS ${CMAKE_SOURCE_DIR}/music/theme.mp3)
  install(FILES music/theme.mp3 DESTINATION ${CMAKE_INSTALL_DATADIR}/shooter-upper/music)
endif()
install(FILES README.md LICENSE DESTINATION ${CMAKE_INSTALL_DOCDIR})

# ---- Debian / Ubuntu package: cmake --build build --target package ----
set(CPACK_GENERATOR "DEB")
set(CPACK_PACKAGE_NAME "shooter-upper")
set(CPACK_PACKAGE_VERSION ${PROJECT_VERSION})
set(CPACK_PACKAGE_CONTACT "Marko Tahvanainen <murtsu@gmail.com>")
set(CPACK_PACKAGE_DESCRIPTION_SUMMARY "Arcade defense shooter with 16-bit looks and 8-bit sound")
set(CPACK_PACKAGE_DESCRIPTION "Fly over a wrapping planet, shoot the aliens and keep your humanoids alive.\n All graphics and sound are generated by code. Keyboard and game controllers.")
set(CPACK_DEBIAN_PACKAGE_SECTION "games")
set(CPACK_DEBIAN_PACKAGE_PRIORITY "optional")
set(CPACK_DEBIAN_FILE_NAME DEB-DEFAULT)
# let dpkg work out the exact library packages the binary needs
set(CPACK_DEBIAN_PACKAGE_SHLIBDEPS ON)
set(CPACK_DEBIAN_PACKAGE_DEPENDS "libsdl2-2.0-0 (>= 2.0.10)")
include(CPack)
